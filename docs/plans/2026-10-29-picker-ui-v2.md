> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Rebuild the picker GUI as a 360px sidebar (search + filters + deck form + actions, no bottom footer), drop the occasion tag filter, and preview each template's first three slides in a select-on-confirm modal.

**Architecture:** `picker.html` is static reshuffle + new modal (no JS-logic contract changes: payload/save chain/validation untouched). `install.ps1` gains a 3-tile capture per template: temp-copy `template.html`, inject a `scrollTop` script (slides stack vertically in a 100vh scroll-snap container — viewport height must NOT change), screenshot each slide at 1920×1080, validate `> 20 KB`, delete legacy single PNGs after success.

**Tech Stack:** Vanilla HTML/CSS/JS, PowerShell 5.1, Chrome headless, git.
**Spec:** `docs/specs/2026-10-29-picker-ui-v2-design.md`

Replace the whole section-3 block (from `# ---------- section 3:` through the `Write-Host "screenshots: ready..."` line) with:

```powershell
# ---------- section 3: first-three-slides captures via Chrome headless ----------
Add-Type -AssemblyName System.Drawing
$ssDir = Join-Path $PSScriptRoot 'screenshots'
New-Item -ItemType Directory -Force -Path $ssDir | Out-Null
$chromeKey0 = Get-ItemProperty 'HKCU:\SOFTWARE\Microsoft\Windows\CurrentVersion\App Paths\chrome.exe' -ErrorAction SilentlyContinue
if (-not $chromeKey0) { $chromeKey0 = Get-ItemProperty 'HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\App Paths\chrome.exe' -ErrorAction SilentlyContinue }
$chrome = if ($chromeKey0) { $chromeKey0.'(default)' } else { $null }
if (-not $chrome -or -not (Test-Path $chrome)) { throw 'Chrome (chrome.exe) not found via App Paths keys (HKCU/HKLM)' }

function Get-ChromeCapture([string]$tmpFile, [string]$pngPath) {
    $null = & $chrome --headless --disable-gpu --hide-scrollbars `
        --default-background-color=FFFFFFFF --virtual-time-budget=15000 `
        '--window-size=1920,1080' "--screenshot=$pngPath" ('file:///' + ($tmpFile -replace '\\','/')) 2>$null
}

$bad = @()
foreach ($t in $templates) {
    $src  = Join-Path $repo ('templates\' + $t.slug + '\template.html')
    if (-not (Test-Path $src)) { $bad += ('no template.html: ' + $t.slug); continue }
    $html = Get-Content $src -Raw -Encoding UTF8
    $tiles = @()
    foreach ($i in 1..3) {
        $png = Join-Path $ssDir ($t.slug + '-' + $i + '.png')
        $upToDate = (Test-Path $png) -and (Get-Item $png).Length -gt 20KB
        if ($upToDate -and -not $Force) { $tiles += $png; continue }
        # inject: kill scroll-snap so scrollTop holds, jump to slide i on load
        $inject = '<style>html,body{margin:0} .slides-container{scroll-snap-type:none !important;' `
            + 'scroll-behavior:auto !important; overflow-y:hidden !important}' `
            + '</style><script>window.addEventListener("load",function(){' `
            + 'var c=document.querySelector(".slides-container"); if(c){c.scrollTop=' `
            + (1080 * ($i - 1)) + ';}});</script>'
        $marker = '</body>'
        $idx = $html.LastIndexOf($marker)
        if ($idx -lt 0) { $bad += ('no </body> to inject: ' + $t.slug); continue }
        $tmp = Join-Path $ssDir ('~' + $t.slug + '-' + $i + '.html')
        Set-Content -Path $tmp -Value ($html.Substring(0, $idx) + $inject + $html.Substring($idx)) -Encoding UTF8
        try {
            Get-ChromeCapture $tmp $png
            for ($w = 0; $w -lt 20; $w++) {
                if ((Test-Path $png) -and (Get-Item $png).Length -gt 20KB) { break }
                Start-Sleep -Milliseconds 500
            }
        } finally {
            Remove-Item $tmp -Force -ErrorAction SilentlyContinue
        }
        if (-not (Test-Path $png)) { $bad += ('capture missing: ' + $t.slug + '-' + $i); continue }
        if ((Get-Item $png).Length -le 20KB) { $bad += ('capture failed: ' + $t.slug + '-' + $i); continue }
        $tiles += $png
    }
    # legacy single-page capture: delete only once the slug's tiles are all fresh
    $legacy = Join-Path $ssDir ($t.slug + '.png')
    if (($tiles.Count -eq 3) -and (Test-Path $legacy)) { Remove-Item $legacy -Force }
}
if ($bad.Count -gt 0) { throw ($bad -join '; ') }
Write-Host "screenshots: 3 tiles per template for $($templates.Count) templates"
```

function confirmSelect(slug) {
    recalled = slug;
    localStorage.setItem('picker.lastTemplate', slug);
    cards(); ribbon(); revalidate();
}
function openShotModal(slug) {
    const t = T.find(x => x.slug === slug);
    $('#modalName').textContent = t.name;
    $('#modalTag').textContent  = t.tagline;
    document.querySelectorAll('[data-modal-shot]').forEach(img => {
        img.src = 'screenshots/' + encodeURIComponent(slug) + '-' + img.dataset.modalShot + '.png';
    });
    const sel = $('#modalSelect');
    sel.onclick = () => { closeShotModal(); confirmSelect(slug); };
    $('#shotModal').classList.add('open');
}
function closeShotModal() { $('#shotModal').classList.remove('open'); }
$('#modalClose').onclick = closeShotModal;
$('#shotModal').addEventListener('click', e => { if (e.target === e.currentTarget) closeShotModal(); });
document.addEventListener('keydown', e => { if (e.key === 'Escape') closeShotModal(); });
